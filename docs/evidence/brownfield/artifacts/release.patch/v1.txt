+## Scale and reliability boundaries
+
+The service is designed for a single region behind a trusted reverse proxy. What holds, and what
+does not, when it runs as several instances:
+
+| Concern | Behaviour | Boundary |
+|---|---|---|
+| Expiry | Enforced on every redirect from the cached `expiresAt`, on every instance | Clock skew between instances shifts it by the skew |
+| Deletion | The deleting instance evicts its cache entry immediately | Other instances may keep redirecting a deleted link until their entry's TTL ends (10 minutes by default); a shared cache or an invalidation broadcast would close this |
+| Rate limiting | Token bucket per client and per instance | N instances allow N times the limit; a shared store (e.g. Redis) is the scale-out answer |
+| Hot links | Cache hits never touch the database; a miss is one indexed query | Click recording adds one batched insert per 500 clicks per instance, not one per redirect |
+| Analytics backpressure | Bounded queue, drops counted | Not durable: use a durable log (e.g. Kafka) if every click must be counted |
+| Management API | `POST`/`GET`/`DELETE` and statistics are unauthenticated; anyone who knows a code can delete it or read its statistics | Must be protected at the gateway (authentication, ownership) before exposure beyond trusted callers |
+| Throughput | No load target is verified in CI | See the load test in `tools/loadtest` and its documented results for one machine |

+package io.github.marishasahay.shortener.click;
+
+import java.util.List;
+import java.util.Locale;
+
+/**
+ * Coarse user-agent family stored with a click instead of the user-agent string, which can
+ * fingerprint a visitor.
+ */
+public enum AgentFamily {
+  BROWSER,
+  BOT,
+  OTHER;
+
+  private static final List<String> BOT_MARKERS = List.of("bot", "crawler", "spider", "slurp",
+      "curl", "wget", "python-requests", "httpclient", "okhttp", "headless");
+  private static final List<String> BROWSER_MARKERS = List.of("mozilla", "chrome", "safari",
+      "firefox", "edg", "opera");
+
+  /**
+   * Classifies a user-agent header.
+   *
+   * @param userAgent the header value, possibly null
+   * @return the family; bots win over browsers because most bots claim to be Mozilla
+   */
+  public static AgentFamily of(String userAgent) {
+    if (userAgent == null || userAgent.isBlank()) {
+      return OTHER;
+    }
+    String ua = userAgent.toLowerCase(Locale.ROOT);
+    if (BOT_MARKERS.stream().anyMatch(ua::contains)) {
+      return BOT;
+    }
+    return BROWSER_MARKERS.stream().anyMatch(ua::contains) ? BROWSER : OTHER;
+  }
+}

+package io.github.marishasahay.shortener.click;
+
+import java.net.URI;
+import java.time.Instant;
+import java.util.Locale;
+
+/**
+ * One successful redirect, reduced to what analytics needs. No IP address, no user-agent string
+ * and no referrer path or query (which can carry personal data) are kept.
+ *
+ * @param code the short code that was followed
+ * @param clickedAt when the redirect was served
+ * @param referrerHost host of the {@code Referer} header, or null
+ * @param agentFamily coarse user-agent family
+ */
+public record ClickEvent(String code, Instant clickedAt, String referrerHost,
+    AgentFamily agentFamily) {
+
+  static final int MAX_HOST_LENGTH = 255;
+
+  /**
+   * Builds an event from raw request headers.
+   *
+   * @param code the short code
+   * @param clickedAt when the redirect was served
+   * @param referer the {@code Referer} header, possibly null
+   * @param userAgent the {@code User-Agent} header, possibly null
+   * @return the reduced event
+   */
+  public static ClickEvent of(String code, Instant clickedAt, String referer, String userAgent) {
+    return new ClickEvent(code, clickedAt, referrerHost(referer), AgentFamily.of(userAgent));
+  }
+
+  /**
+   * Host part of a referrer URL.
+   *
+   * @param referer header value
+   * @return lower-case host, or null if absent or not an absolute http(s) URL
+   */
+  static String referrerHost(String referer) {
+    if (referer == null || referer.isBlank()) {
+      return null;
+    }
+    try {
+      URI uri = URI.create(referer.strip());
+      String scheme = uri.getScheme();
+      String host = uri.getHost();
+      if (host == null || scheme == null
+          || !("http".equalsIgnoreCase(scheme) || "https".equalsIgnoreCase(scheme))) {
+        return null;
+      }
+      String normalised = host.toLowerCase(Locale.ROOT);
+      return normalised.length() > MAX_HOST_LENGTH ? null : normalised;
+    } catch (IllegalArgumentException e) {
+      return null;
+    }
+  }
+}

+
+import io.github.marishasahay.shortener.config.ShortenerProperties;
+import io.micrometer.core.instrument.Counter;
+import io.micrometer.core.instrument.MeterRegistry;

+import java.util.List;
+import java.util.concurrent.ArrayBlockingQueue;
+import java.util.concurrent.BlockingQueue;
+import org.slf4j.Logger;

+/**
+ * Records clicks off the redirect path, best effort. The redirect only offers the click to a
+ * bounded in-memory queue (never blocks, never touches the database); a scheduled flusher writes
+ * queued clicks in batches. Clicks are lost, and counted, when the queue is full, when a batch
+ * insert fails, or when the process dies before a flush; a graceful shutdown flushes first.
+ * Counters: {@code shortener.clicks.recorded}, {@code .dropped} and {@code .failed}.
+ */
+@Component
+public class ClickRecorder {
+
+  private static final Logger LOG = LoggerFactory.getLogger(ClickRecorder.class);
+
+  private final BlockingQueue<ClickEvent> queue;
+  private final int batchSize;
+  private final ClickRepository repository;
+  private final Clock clock;
+  private final Counter recorded;
+  private final Counter dropped;
+  private final Counter failed;
+
+  /**
+   * Creates the recorder.
+   *
+   * @param properties queue capacity and batch size
+   * @param repository click persistence
+   * @param clock time source
+   * @param meterRegistry where the counters are published
+   */
+  public ClickRecorder(ShortenerProperties properties, ClickRepository repository, Clock clock,
+      MeterRegistry meterRegistry) {
+    ShortenerProperties.Clicks settings = properties.clicks();
+    this.queue = new ArrayBlockingQueue<>(settings.queueCapacity());
+    this.batchSize = settings.batchSize();
+    this.repository = repository;
+    this.clock = clock;
+    this.recorded = meterRegistry.counter("shortener.clicks.recorded");
+    this.dropped = meterRegistry.counter("shortener.clicks.dropped");
+    this.failed = meterRegistry.counter("shortener.clicks.failed");
+    meterRegistry.gaugeCollectionSize("shortener.clicks.queued", List.of(), queue);
+  }
+
+  /**
+   * Offers a click for recording. Constant time; never blocks the caller.
+   *
+   * @param code the short code that was followed
+   * @param referer the {@code Referer} header, possibly null
+   * @param userAgent the {@code User-Agent} header, possibly null
+   * @return whether the click was queued (false: queue full, click dropped and counted)
+   */
+  public boolean record(String code, String referer, String userAgent) {
+    if (queue.offer(ClickEvent.of(code, clock.instant(), referer, userAgent))) {
+      return true;
+    }
+    dropped.increment();
+    return false;
+  }
+
+  /**
+   * Writes every queued click, in batches. Called on a fixed delay and at shutdown.
+   *
+   * @return number of clicks written
+   */
+  public synchronized int flush() {
+    int written = 0;
+    List<ClickEvent> batch = new ArrayList<>(batchSize);
+    while (queue.drainTo(batch, batchSize) > 0) {
+      try {
+        repository.insertBatch(batch);
+        recorded.increment(batch.size());
+        written += batch.size();
+      } catch (RuntimeException e) {
+        failed.increment(batch.size());
+        LOG.warn("Lost {} clicks: batch insert failed ({})", batch.size(), e.getMessage());
+      }
+      batch.clear();
+    }
+    return written;
+  }
+
+  /** Flushes what is queued before the application stops. */
+  @PreDestroy
+  public void flushOnShutdown() {
+    flush();
+  }
+
+  int queued() {
+    return queue.size();
+  }
+}
